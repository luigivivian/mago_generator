  <action>
Create `lib/editor/snap.ts` exporting:
```ts
export interface SnapTarget { frame: number; label?: string }
export interface SnapResult { frame: number; snapped: boolean; targetLabel?: string }

export function snapFrame(
  candidate: number,
  targets: SnapTarget[],
  pixelsPerFrame: number,
  thresholdPx = 6,
): SnapResult
```

Logic: convert thresholdPx to frames (`thresholdPx / pixelsPerFrame`),
find the nearest target within that distance, return `{ frame: nearest,
snapped: true }` or `{ frame: candidate, snapped: false }` if none.
Pure function — no React, no DOM.

Add `export { snapFrame } from "./snap"` to barrel `index.ts`.

   ```ts
   const snapTargets = useMemo<SnapTarget[]>(() => {
     const targets: SnapTarget[] = [{ frame: playheadFrame, label: "playhead" }];
     let off = 0;
     for (const s of scenes) {
       targets.push({ frame: off, label: `scene-${s.index}-start` });
       off += s.durationInFrames;
       targets.push({ frame: off, label: `scene-${s.index}-end` });
     }
     return targets;
   ```
