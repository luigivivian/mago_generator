```python
def export_takes_multi_format(
    composed_video_path: str,
    take_video_paths: list[str],
    output_dir: str,
    formats: list[str],
) -> dict:
    """Export final composed video in multiple formats AND individual takes.
    
    Args:
        composed_video_path: Path to the final composed video (from compose_takes).
        take_video_paths: Paths to individual take videos for per-take export.
        output_dir: Base output directory.
        formats: List of aspect ratios: "9:16", "16:9", "1:1".
    
    Returns:
        {
          "final": {"9:16": path, "16:9": path, "1:1": path},
          "takes": [{"9:16": path, ...}, ...],
          "thumbnail": path,
        }
    """
    import os
    
    os.makedirs(output_dir, exist_ok=True)
    takes_dir = os.path.join(output_dir, "takes")
    os.makedirs(takes_dir, exist_ok=True)
    
    # Export final composed video in all formats
    final_exports = export_all_formats(composed_video_path, output_dir, formats)
    
    # Export each take in all formats
    takes_exports = []
    for i, take_path in enumerate(take_video_paths):
        take_out_dir = os.path.join(takes_dir, f"take_{i}")
        os.makedirs(take_out_dir, exist_ok=True)
        take_exports = export_all_formats(take_path, take_out_dir, formats)
        takes_exports.append(take_exports)
    
    # Generate thumbnail from the first frame of composed video
    thumbnail_path = os.path.join(output_dir, "thumbnail.jpg")
    generate_thumbnail(composed_video_path, thumbnail_path)
    
    return {
        "final": final_exports,
        "takes": takes_exports,
        "thumbnail": thumbnail_path,
    }


def generate_thumbnail(
    video_path: str,
    output_path: str,
    timestamp_sec: float = 0.5,
) -> str:
    """Extract a thumbnail JPG from a video at the given timestamp.
    
    Uses ffmpeg -ss for fast seek.
    """
    import subprocess
    cmd = [
        "ffmpeg", "-y",
        "-ss", str(timestamp_sec),
        "-i", video_path,
        "-frames:v", "1",
        "-q:v", "2",
        output_path,
    ]
    subprocess.run(cmd, check=True, capture_output=True, timeout=30)
    return output_path
```

```python
async def run_v2_pipeline(
    self,
    image_urls: list[str],
    category: str,
    product_name: str,
    takes: list[dict] | None = None,
    output_formats: list[str] | None = None,
    audio_mode: str = "sfx",
    aspect_ratio: str = "9:16",
) -> dict:
    """End-to-end v2 cinematic pipeline.
    
    Steps:
    1. If takes is None: generate storyboard via scene_composer.generate_storyboard
    2. Download images from image_urls to local paths
    3. Generate videos via run_step_generation_v2 (Kling multi-shot or per-take)
    4. Auto-select SFX per take (if audio_mode == "sfx")
    5. Compose final audio (ambient + SFX hits) via compose_take_audio
    6. Compose final video via compose_takes (per-transition xfade)
    7. Mux audio + video into composed.mp4
    8. Export multi-format + individual takes + thumbnail
    
    Returns dict with output paths and metadata.
    """
    import os
    import tempfile
    import httpx
    from src.product_studio.scene_composer import generate_storyboard
    from src.product_studio.take_composer import compose_takes, compose_take_audio, calculate_sfx_offsets
    from src.product_studio.sfx_library import auto_select_sfx_for_product, resolve_sfx_path, get_sfx_by_id
    from src.product_studio.format_exporter import export_takes_multi_format
    
    output_formats = output_formats or ["9:16"]
    job_dir = self._make_job_dir(product_name)
    
    # Download images to local paths for scene generation
    local_image_paths = []
    async with httpx.AsyncClient(timeout=60.0) as hc:
        for i, url in enumerate(image_urls):
            resp = await hc.get(url)
            resp.raise_for_status()
            local = os.path.join(job_dir, f"ref_{i}.jpg")
            with open(local, "wb") as fp:
                fp.write(resp.content)
            local_image_paths.append(local)
    
    # Step 1: Generate storyboard if no takes provided
    if not takes:
        storyboard = await generate_storyboard(
            image_paths=local_image_paths,
            category=category,
            product_name=product_name,
            num_takes=4,
        )
        takes = [s["take_config"] for s in storyboard]
    
    # Step 2: Generate videos via Kling
    gen_result = await self.run_step_generation_v2(
        image_urls=image_urls,
        takes=takes,
        job_dir=job_dir,
        product_name=product_name,
        aspect_ratio=aspect_ratio,
    )
    video_paths = gen_result["video_paths"]
    
    # If single multi_shot video was returned, split it into per-take paths for composition.
    # For simplicity in v2.0: if mode == multi_shot, skip compose_takes and use the single video.
    if gen_result["mode"] == "multi_shot":
        composed_video = video_paths[0]
        take_video_paths = [composed_video]  # No individual takes to export
    else:
        # Per-take mode: compose with xfade transitions
        transition_types = [t.get("transition_type", "dissolve") for t in takes[:-1]]
        composed_video = os.path.join(job_dir, "composed_video.mp4")
        compose_takes(
            video_paths=video_paths,
            transition_types=transition_types,
            output_path=composed_video,
            transition_duration=0.5,
        )
        take_video_paths = video_paths
    
    # Step 3: Audio composition (only if audio_mode == "sfx")
    if audio_mode == "sfx":
        sfx_selection = auto_select_sfx_for_product(category)
        ambient_path = resolve_sfx_path(sfx_selection["ambient_id"])
        
        take_durations = [float(t.get("duration", 5)) for t in takes]
        offsets = calculate_sfx_offsets(take_durations, transition_duration=0.5)
        
        sfx_entries = []
        hit_ids = sfx_selection["hit_ids"]
        for i, offset in enumerate(offsets):
            sfx_id = hit_ids[i % len(hit_ids)] if hit_ids else None
            path = resolve_sfx_path(sfx_id) if sfx_id else None
            if path:
                sfx_entries.append({"path": path, "offset_sec": offset, "volume": 0.8})
        
        total_duration = sum(take_durations)
        audio_path = os.path.join(job_dir, "composed_audio.m4a")
        compose_take_audio(
            ambient_path=ambient_path,
            sfx_entries=sfx_entries,
            output_path=audio_path,
            total_duration=total_duration,
        )
        
        # Mux audio + video via ffmpeg
        import subprocess
        muxed_path = os.path.join(job_dir, "composed_muxed.mp4")
        subprocess.run([
            "ffmpeg", "-y",
            "-i", composed_video,
            "-i", audio_path,
            "-c:v", "copy",
            "-c:a", "aac",
            "-map", "0:v:0",
            "-map", "1:a:0",
            "-shortest",
            muxed_path,
        ], check=True, capture_output=True, timeout=120)
        composed_video = muxed_path
    
    # Step 4: Export multi-format + individual takes + thumbnail
    exports = export_takes_multi_format(
        composed_video_path=composed_video,
        take_video_paths=take_video_paths,
        output_dir=os.path.join(job_dir, "exports"),
        formats=output_formats,
    )
    
    return {
        "job_dir": job_dir,
        "composed_video": composed_video,
        "exports": exports,
        "takes": takes,
        "cost_usd": gen_result["cost_usd"],
        "mode": gen_result["mode"],
    }
```

```python
    """Create a v2 cinematic multi-scene ad job.

        takes=request.takes,
        storyboard=[],
    )


async def _run_v2_pipeline_task(job_id: str, user_id: str, request_dict: dict):
    """Background task: run the v2 pipeline and update DB with results."""
    from src.database.session import SessionLocal
    from src.database.models import ProductAdJob
    from src.product_studio.pipeline import ProductAdPipeline
    
    db = SessionLocal()
    try:
        job = db.query(ProductAdJob).filter(ProductAdJob.id == job_id).first()
        if not job:
            return
        job.status = "processing"
        db.commit()
        
        pipeline = ProductAdPipeline()
        result = await pipeline.run_v2_pipeline(
            image_urls=request_dict["image_urls"],
            category=request_dict["category"],
            product_name=request_dict["product_name"],
            takes=request_dict.get("takes") or None,
            output_formats=request_dict.get("output_formats") or ["9:16"],
            audio_mode=request_dict.get("audio_mode", "sfx"),
        )
        
        job.status = "complete"
        job.outputs = {
            "composed_video": result["composed_video"],
            "exports": result["exports"],
            "mode": result["mode"],
        }
        job.takes_config = result["takes"]
        job.cost_brl = float(result["cost_usd"]) * 5.0  # rough USD->BRL
        db.commit()
    except Exception as e:
        import logging
        logging.getLogger("clip-flow.ads.v2").exception("v2 pipeline failed: %s", e)
        # None-guard: `job` may be None if the initial .filter(...).first() returned None
        # (e.g., race with deletion), or may be stale after commit errors.
        if job is not None:
            try:
                job.status = "failed"
                job.error_message = str(e)[:500]
                db.commit()
            except Exception:
                db.rollback()
        raise
    finally:
        db.close()
```
