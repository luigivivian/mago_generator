export interface EditorScene {
  id: string;
  index: number;
  clipUrl?: string;
  imgUrl?: string;
  durationInFrames: number;
  narration: string;
  voiceConfig: { voice: string; speed: number };
  transition: { type: "fade" | "slide" | "wipe" | "flip" | "none"; durationFrames: number };

}

export interface EditorSubtitle {
  id: string;
  text: string;
  startFrame: number;
  endFrame: number;
  position: { x: number; y: number };
  style: { fontSize: number; fontFamily: string; color: string; shadowColor: string; shadowSize: number };
}

export interface EditorAudioItem {
  id: string;
  audioUrl: string;
  from: number;
  durationInFrames: number;
}

export interface EditorTrack {
  type: "video" | "audio" | "subtitle";
  items: EditorScene[] | EditorAudioItem[] | EditorSubtitle[];
}

- If only `scene.imgUrl`: render `<Img src={scene.imgUrl} style={{ width: "100%", height: "100%", objectFit: "cover" }} />`
- Wrap in `<AbsoluteFill style={{ backgroundColor: "#000" }}>`
- Import `Video` from `@remotion/media` and `Img` from `remotion`
- NEVER use useState or useEffect inside Remotion compositions (per RESEARCH anti-patterns)

2. Create `memelab/src/remotion/components/SubtitleOverlay.tsx`:
- Props: `{ subtitles: EditorSubtitle[], fps: number }`
- Uses `useCurrentFrame()` from remotion to determine which subtitles are visible at current frame
- For each visible subtitle: render positioned `<div>` with inline styles from subtitle.style (fontSize, fontFamily, color, textShadow from shadowColor+shadowSize)
- Position via `{ position: "absolute", left: `${sub.position.x}%`, top: `${sub.position.y}%` }`
- Wrap in `<AbsoluteFill>` with `pointerEvents: "none"`

3. Create `memelab/src/remotion/ReelComposition.tsx`:
- Props: `{ tracks: EditorTrack[] }` — receives all data via inputProps
- Extract video, audio, subtitle tracks from tracks array
- Render video scenes using `<TransitionSeries>` from `@remotion/transitions`:
  - For each scene: `<TransitionSeries.Sequence durationInFrames={scene.durationInFrames}><Scene scene={scene} /></TransitionSeries.Sequence>`
  - Between scenes (if scene.transition.type !== "none" AND not last scene): `<TransitionSeries.Transition presentation={getPresentation(scene.transition.type)} timing={linearTiming({ durationInFrames: scene.transition.durationFrames })} />`
- Map transition types to presentations: fade() from "@remotion/transitions/fade", slide() from "@remotion/transitions/slide", wipe() from "@remotion/transitions/wipe", flip() from "@remotion/transitions/flip"
- Render audio items: each `<Sequence from={item.from} durationInFrames={item.durationInFrames}><Audio src={item.audioUrl} /></Sequence>`
- Render subtitle overlay: `<SubtitleOverlay subtitles={subtitleItems} fps={fps} />`
- Use `useVideoConfig()` for fps
- Wrap everything in `<AbsoluteFill style={{ backgroundColor: "#000" }}>`
